"""
jvm_boot_glue.py
────────────────
Bridges the memory-resident JVM into the JPype runtime and launches the
Minecraft main class entirely in-process.

Architecture — three-stage boot
================================

Stage 1 · IAT Hook  (before jpype.startJVM)
    Patch two kernel32 entries in _jpype.pyd's Import Address Table:
      · LoadLibraryW   → _hook_load_library_w
      · GetProcAddress → _hook_get_proc_address
    Only _jpype.pyd's private IAT copy is modified.

Stage 2 · Transparent JVM Creation  (inside jpype.startJVM)
    jpype.startJVM() calls _jpype.startup(jvmpath, full_args, …)
    → C++: LoadLibraryW(jvm.dll)              ← HOOKED → fake HINSTANCE
    → C++: GetProcAddress(fake, "JNI_CreateJavaVM") ← HOOKED → trampoline
    → C++: trampoline(pvm, penv, jniArgs)     → delegates to in-memory jvm.dll
    → C++: m_Running = true
    → C++/Python: initializeResources()        ← runs normally

Stage 3 · In-Process Game Launch  (after jpype.startJVM)
    IAT is restored.  JPype is fully operational.
    jpype.JClass(main_class).main(game_args[:])

JPype 1.0+ compliance
=====================
· No JArray / JObject usage
· No manual thread attachment
· No forced shutdown
· No @JImplements — lambdas used for SAM types
· Dynamic class loading used post-JVM-start
"""

def _flush_logging() -> None:
    for handler in logging.getLogger().handlers:
        try:
            handler.flush()
        except Exception:
            pass


def _enable_native_crash_log() -> None:
    global _CRASH_LOG_FILE
    if _CRASH_LOG_FILE is not None:
        return
    try:
        crash_dir = Path.cwd() / "logs"
        crash_dir.mkdir(parents=True, exist_ok=True)
        _CRASH_LOG_FILE = (crash_dir / "python_native_crash.log").open("a", encoding="utf-8")
        _CRASH_LOG_FILE.write("\n=== jvm_boot_glue native crash capture enabled ===\n")
        _CRASH_LOG_FILE.flush()
        faulthandler.enable(file=_CRASH_LOG_FILE, all_threads=True)
    except Exception as exc:
        log.warning("Could not enable native crash log: %s", exc)


def _wait_for_java_non_daemon_threads() -> None:
    """Keep Python alive until all Java non-daemon threads finish."""
    import jpype
    Thread = jpype.JClass("java.lang.Thread")
    current = Thread.currentThread()
    log.info("Minecraft main() returned; waiting for Java non-daemon threads")
    while True:
        threads = list(Thread.getAllStackTraces().keySet().toArray())
        live = [t for t in threads if t.isAlive() and not t.isDaemon() and t != current]
        if not live:
            log.info("No Java non-daemon threads remain")
            return
        names = ", ".join(str(t.getName()) for t in live[:8])
        log.debug("Waiting on Java threads: %s", names)
        time.sleep(1.0)


def _safe_subprocess_cwd(path) -> str:
    """
    Return a cwd string safe for subprocess — never a UNC path.

    Windows cannot set the working directory to a UNC path (\\server\share\...)
    in a child process.  Fall back to the system temp directory when the
    requested path is UNC.
    """
    if path is None:
        return tempfile.gettempdir()
    p = str(path)
    if p.startswith("\\\\"):
        return tempfile.gettempdir()
    try:
        resolved = str(Path(p).resolve())
        return resolved if not resolved.startswith("\\\\") else tempfile.gettempdir()
    except Exception:
        return tempfile.gettempdir()


def _without_local_launcher_path(paths: list[str]) -> list[str]:
    """
    Remove this launcher's scripts/ directory from a sys.path / PYTHONPATH list.

    scripts/portablemc.py is the Flask UI launcher.  If scripts/ is on
    sys.path it shadows the installed *portablemc* package and causes
    ``python -m portablemc`` to run the guard stub instead of the real CLI.
    """
    cleaned: list[str] = []
    local = str(_LOCAL_SCRIPT_DIR).casefold()
    for entry in paths:
        if not entry:
            cleaned.append(entry)
            continue
        try:
            if str(Path(entry).resolve()).casefold() == local:
                continue
        except OSError:
            pass
        cleaned.append(entry)
    return cleaned


class _CleanImportPath:
    """Context manager: temporarily hide scripts/ from sys.path."""

    def __enter__(self) -> None:
        self._old = list(sys.path)
        sys.path[:] = _without_local_launcher_path(sys.path)

    def __exit__(self, *_) -> None:
        sys.path[:] = self._old

def _parse_portablemc_dry_output(raw: str) -> tuple | None:
    """
    Try every known portablemc output format.

    Returns (jvm_flags, classpath, main_class, game_args) or None.

    Strategies tried in order
    ─────────────────────────
    1. JSON lines — portablemc 5.x Rust ``--output machine`` and portablemc
       4.x ``--output json-line``.
    2. "Arguments:" indented text block — portablemc 4.x human-color output.
    3. Tab-delimited machine format — portablemc 5.x Rust machine events.
    """
    # ── Strategy 1: JSON lines ────────────────────────────────────────────
    for line in raw.splitlines():
        line = line.strip()
        if not line.startswith("{"):
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError:
            continue

        # portablemc 5.x Rust: {"event": "jvm_args", "args": [...]}
        if obj.get("event") == "jvm_args":
            args = obj.get("args") or []
            if isinstance(args, list) and len(args) >= 2:
                log.debug("Parsed via JSON event=jvm_args (%d args)", len(args))
                return _PortableMCParse.split_flat(args)

        # portablemc 4.x: various game-launch events
        for evt in ("game_launch", "game_start", "jvm_start", "jvm"):
            if obj.get("event") == evt:
                jvm = obj.get("jvm_args") or obj.get("args") or []
                mc = obj.get("main_class", "")
                ga = obj.get("game_args") or []
                if isinstance(jvm, list) and jvm:
                    log.debug("Parsed via JSON event=%s", evt)
                    return _PortableMCParse.extract(jvm, ga, mc)

        # Generic: any JSON object that contains a list with "-cp"
        for key in ("args", "command", "jvm_args", "jvm"):
            val = obj.get(key)
            if isinstance(val, list) and "-cp" in val:
                log.debug("Parsed via JSON key=%s with -cp", key)
                return _PortableMCParse.split_flat(val)

    # ── Strategy 2: "Arguments:" text block ──────────────────────────────
    args = _parse_arguments_block(raw)
    if args is not None:
        log.debug("Parsed via 'Arguments:' text block (%d args)", len(args))
        return _PortableMCParse.split_flat(args)

    # ── Strategy 3: tab-delimited Rust machine format ─────────────────────
    args = _parse_tab_machine(raw)
    if args is not None:
        log.debug("Parsed via tab-machine format (%d args)", len(args))
        return _PortableMCParse.split_flat(args)

    return None


def _parse_arguments_block(raw: str) -> list[str] | None:
    """
    Extract the arg list from portablemc's human-readable "Arguments:" block.

    portablemc 4.x with ``--output human-color`` (or default) prints:

        [ INFO ] Arguments:
                   -Djava.home=/path/to/jdk
                   -Djava.library.path=/path/to/natives
                   -cp
                   jar1.jar;jar2.jar
                   net.fabricmc.loader.impl.launch.knot.KnotClient
                   --username Player
                   ...

    Each argument is on its own line, indented.
    """
    lines = raw.splitlines()
    in_block = False
    args: list[str] = []

    for line in lines:
        # Strip ANSI escape codes
        clean = re.sub(r"\x1b\[[0-9;]*m", "", line)
        stripped = clean.strip()

        if not in_block:
            if re.search(r"\barguments?\b\s*:", stripped, re.IGNORECASE):
                in_block = True
            continue

        if not stripped:
            if args:
                break  # blank line ends the block
            continue

        # Stop at a new top-level section ([ OK ], [ INFO ], etc.)
        if re.match(r"^\[", stripped) and re.search(r"\]\s+\S", stripped[:30]):
            break

        # Strip common leading decoration characters
        arg = stripped.lstrip("| \t")
        if arg:
            args.append(arg)

    # Need at least a -cp flag and a classpath to be a valid block
    if len(args) < 3 or "-cp" not in args:
        return None
    return args


def _parse_tab_machine(raw: str) -> list[str] | None:
    """
    Parse portablemc 5.x Rust tab-delimited machine output.

    Lines look like:  ``jvm_args\\tadditional\\t-Djava.home=/path``
    """
    args: list[str] = []
    for line in raw.splitlines():
        parts = line.split("\t")
        if (
            len(parts) >= 3
            and parts[0].strip() == "jvm_args"
            and parts[1].strip() in ("additional", "args", "info")
        ):
            val = parts[2].strip()
            if val and val.lower() not in ("arguments:", "arguments"):
                args.append(val)
    return args if len(args) >= 3 else None


class _PortableMCParse:
    """Static helpers for splitting portablemc arg lists."""

    @staticmethod
    def split_flat(args: list[str]) -> tuple:
        """Split a flat [jvm… main_class game…] list into four components."""
        return _PortableMCParse.extract(args, [], "")

    @staticmethod
    def extract(
        jvm_args: list[str],
        game_args: list[str],
        main_class: str,
    ) -> tuple[list[str], list[str], str, list[str]]:
        """Separate classpath from JVM flags; identify main class."""
        classpath: list[str] = []
        pure_flags: list[str] = []
        skip_next = False

        for i, arg in enumerate(jvm_args):
            if skip_next:
                skip_next = False
                continue
            if arg in ("-cp", "-classpath"):
                if i + 1 < len(jvm_args):
                    classpath = jvm_args[i + 1].split(os.pathsep)
                    skip_next = True
            elif arg.startswith("-Djava.class.path="):
                classpath = arg[len("-Djava.class.path="):].split(os.pathsep)
            else:
                pure_flags.append(arg)

        # If the last token looks like a main class (has dots, no leading dash)
        # and game_args is empty, it was baked into jvm_args by portablemc.
        if not game_args and pure_flags:
            last = pure_flags[-1]
            if "." in last and not last.startswith("-") and not last.endswith(".jar"):
                if not main_class:
                    main_class = last
                pure_flags = pure_flags[:-1]
                # Everything after the main class index belongs to game_args.
                # Re-scan: find where main_class lives in original jvm_args.
                try:
                    mc_idx = jvm_args.index(last)
                    game_args = list(jvm_args[mc_idx + 1:])
                    # Rebuild pure_flags without the main class and game args
                    raw_before_mc = jvm_args[:mc_idx]
                    classpath = []
                    pure_flags = []
                    skip_next2 = False
                    for i, arg in enumerate(raw_before_mc):
                        if skip_next2:
                            skip_next2 = False
                            continue
                        if arg in ("-cp", "-classpath"):
                            if i + 1 < len(raw_before_mc):
                                classpath = raw_before_mc[i + 1].split(os.pathsep)
                                skip_next2 = True
                        elif arg.startswith("-Djava.class.path="):
                            classpath = arg[len("-Djava.class.path="):].split(os.pathsep)
                        else:
                            pure_flags.append(arg)
                except ValueError:
                    pass

        log.info(
            "Extracted: %d JVM flags, %d classpath entries, main=%r, %d game args",
            len(pure_flags), len(classpath), main_class, len(game_args),
        )
        return pure_flags, classpath, main_class, game_args

# § 3  IAT Patcher

class _IATEntry:
    """One patchable slot in a loaded module's Import Address Table."""

    _k32 = ctypes.windll.kernel32

    def __init__(self, abs_addr: int) -> None:
        self._addr = abs_addr
        self._original: int = ctypes.c_void_p.from_address(abs_addr).value

    @property
    def original(self) -> int:
        return self._original

    def patch(self, hook_fn_ptr: int) -> None:
        vp = self._k32.VirtualProtect
        vp.argtypes = [
            ctypes.c_void_p, ctypes.c_size_t,
            ctypes.wintypes.DWORD, ctypes.POINTER(ctypes.wintypes.DWORD),
        ]
        vp.restype = ctypes.wintypes.BOOL
        old = ctypes.wintypes.DWORD(0)
        if not vp(ctypes.c_void_p(self._addr), _PTR_SIZE, 0x04, ctypes.byref(old)):
            raise OSError(
                f"VirtualProtect(rw) failed at 0x{self._addr:016x}: "
                f"error {self._k32.GetLastError()}"
            )
        ctypes.c_void_p.from_address(self._addr).value = hook_fn_ptr
        dummy = ctypes.wintypes.DWORD(0)
        vp(ctypes.c_void_p(self._addr), _PTR_SIZE, old, ctypes.byref(dummy))
        log.debug("IAT 0x%016x ← 0x%016x", self._addr, hook_fn_ptr)

    def restore(self) -> None:
        self.patch(self._original)


def _find_iat_entry(
    module_base: int,
    target_dll: str,
    target_func: str,
    module_path: str | None = None,
) -> _IATEntry:
    """
    Walk a loaded module's in-memory PE headers to locate an IAT slot.

    Tries pefile first (more reliable), then falls back to manual PE walk.
    Only handles x64 PE32+.
    """
    target_dll_upper = target_dll.upper()
    target_func_bytes = target_func.encode("ascii")

    # ── pefile fast path ──────────────────────────────────────────────────
    if module_path:
        try:
            from pythonmemorymodule import pefile as _pefile  # type: ignore[import]
            pe = _pefile.PE(module_path, fast_load=False)
            image_base = pe.OPTIONAL_HEADER.ImageBase
            for entry in getattr(pe, "DIRECTORY_ENTRY_IMPORT", []):
                dll_name = entry.dll.decode("ascii", errors="replace").upper()
                if dll_name != target_dll_upper:
                    continue
                for imp in entry.imports:
                    if imp.name == target_func_bytes:
                        slot_addr = module_base + (imp.address - image_base)
                        pe.close()
                        return _IATEntry(slot_addr)
            pe.close()
        except Exception as exc:
            log.debug("pefile IAT lookup skipped: %s", exc)

    # ── manual PE walk ────────────────────────────────────────────────────
    dos_magic = ctypes.c_uint16.from_address(module_base).value
    if dos_magic != 0x5A4D:
        raise RuntimeError(f"No MZ header at 0x{module_base:x}")
    e_lfanew = ctypes.c_int32.from_address(module_base + 0x3C).value
    nt_base = module_base + e_lfanew
    if ctypes.c_uint32.from_address(nt_base).value != 0x00004550:
        raise RuntimeError("Not a PE file")
    opt_base = nt_base + 4 + 20
    if ctypes.c_uint16.from_address(opt_base).value != 0x20B:
        raise RuntimeError("Expected PE32+ (0x20B)")

    import_dir_va   = ctypes.c_uint32.from_address(opt_base + 112 + 8).value
    import_dir_size = ctypes.c_uint32.from_address(opt_base + 112 + 12).value
    if not import_dir_va or not import_dir_size:
        raise RuntimeError("Module has no import directory")

    desc_addr = module_base + import_dir_va
    while True:
        orig_ft = ctypes.c_uint32.from_address(desc_addr     ).value
        name_rv = ctypes.c_uint32.from_address(desc_addr + 12).value
        first_t = ctypes.c_uint32.from_address(desc_addr + 16).value
        if not orig_ft and not name_rv and not first_t:
            break
        dll_name = ctypes.string_at(module_base + name_rv).decode("ascii", errors="replace").upper()
        if dll_name == target_dll_upper:
            oft_base = module_base + orig_ft
            ft_base  = module_base + first_t
            i = 0
            while True:
                thunk = ctypes.c_uint64.from_address(oft_base + i * _PTR_SIZE).value
                if not thunk:
                    break
                if not (thunk & (1 << 63)):
                    ibn = module_base + (thunk & 0x7FFFFFFFFFFFFFFF)
                    fname = ctypes.string_at(ibn + 2)
                    if fname == target_func_bytes:
                        return _IATEntry(ft_base + i * _PTR_SIZE)
                i += 1
        desc_addr += 20

    raise RuntimeError(
        f"IAT entry {target_dll}!{target_func} not found in module at 0x{module_base:016x}"
    )

# § 4  JvmBootGlue — central orchestrator
# ══════════════════════════════════════════════════════════════════════════════

class JvmBootGlue:
    """
    Bridges the memory-resident JVM into JPype and launches Minecraft in-process.
    """

    def __init__(
        self,
        loader: "JvmMemoryLoader",
        jvm_flags: list[str],
        classpath: list[str],
        main_class: str,
        game_args: list[str],
        debug: bool = False,
    ) -> None:
        self._loader     = loader
        self._jvm_flags  = jvm_flags
        self._classpath  = classpath
        self._main_class = main_class
        self._game_args  = game_args
        self._debug      = debug
        self._fake_hmod: int = loader.jvm_codebase()

        # IAT entries (populated by _patch_iat)
        self._iat_llw:    _IATEntry | None = None
        self._iat_gpa:    _IATEntry | None = None
        self._iat_gmfa:   _IATEntry | None = None
        self._iat_jvm_gmfw:  _IATEntry | None = None
        self._iat_jvm_lla:   _IATEntry | None = None
        self._iat_jvm_llexw: _IATEntry | None = None
        self._iat_jvm_gpa:   _IATEntry | None = None
        self._iat_jvm_fl:    _IATEntry | None = None

        # ctypes callbacks — held alive to prevent GC
        self._hook_llw:    _LoadLibraryW_t | None   = None
        self._hook_gpa:    _GetProcAddress_t | None = None
        self._hook_gmfa:   _GetModuleFileNameA_t | None = None
        self._hook_gmfw:   _GetModuleFileNameW_t | None = None
        self._hook_jvm_lla:   _LoadLibraryA_t | None   = None
        self._hook_jvm_llexw: _LoadLibraryExW_t | None = None
        self._hook_jvm_gpa:   _GetProcAddress_t | None = None
        self._hook_jvm_fl:    _FreeLibrary_t | None    = None
        self._trampoline: _JNI_CreateJavaVM_t | None   = None

    # ── § 4.1  build hook callbacks ───────────────────────────────────────

    def _build_hooks(self) -> None:
        fake = self._fake_hmod
        loader = self._loader
        real_llw = _LoadLibraryW_t(self._iat_llw.original)

        # LoadLibraryW: redirect jvm.dll to our fake handle
        def _h_llw(path: str | None) -> int:
            if path and path.lower().endswith("jvm.dll"):
                log.debug("IAT hook LoadLibraryW(%r) → fake 0x%x", path, fake)
                return fake
            return real_llw(path)
        self._hook_llw = _LoadLibraryW_t(_h_llw)

        # JNI_CreateJavaVM trampoline: forward JPype's args to our in-memory impl
        _real_create = loader.jni_create_java_vm

        def _h_create(pvm, penv, args) -> int:
            log.info("Trampoline: forwarding CreateJavaVM to in-memory jvm.dll")
            rc = _real_create(pvm, penv, args)
            if rc == JNI_OK:
                log.info("  JavaVM* = 0x%016x", pvm[0] if pvm[0] else 0)
                log.info("  JNIEnv* = 0x%016x", penv[0] if penv[0] else 0)
            else:
                log.error("  JNI_CreateJavaVM returned %d", rc)
            return rc
        self._trampoline = _JNI_CreateJavaVM_t(_h_create)

        # GetProcAddress: dispatch to our function table for the fake handle
        trampoline_addr   = ctypes.cast(self._trampoline, ctypes.c_void_p).value
        get_created_addr  = ctypes.cast(loader.jni_get_created_jvms, ctypes.c_void_p).value
        get_initargs_addr = ctypes.cast(loader.jni_get_default_init_args, ctypes.c_void_p).value
        _dispatch: dict[bytes, int] = {
            b"JNI_CreateJavaVM":            trampoline_addr,
            b"JNI_GetCreatedJavaVMs":       get_created_addr,
            b"JNI_GetDefaultJavaVMInitArgs": get_initargs_addr,
        }
        real_gpa = _GetProcAddress_t(self._iat_gpa.original)

        def _h_gpa(hmod: int, name: bytes | None) -> int:
            if hmod == fake and name:
                addr = _dispatch.get(name)
                if addr is not None:
                    log.debug("IAT GetProcAddress(fake, %r) → 0x%x", name, addr)
                    return addr
                log.warning("IAT GetProcAddress(fake, %r) — unknown export → 0", name)
                return 0
            return real_gpa(hmod, name)
        self._hook_gpa = _GetProcAddress_t(_h_gpa)

    def _patch_jvm_iat(self) -> None:
        """
        Patch the IAT of the in-memory jvm.dll itself so that HotSpot's
        internal library-load and self-identification calls work correctly
        against our VirtualAlloc-based image.
        """
        jvm_path = str(self._loader.jdk_bin / "server" / "jvm.dll")
        jvm_path_bytes = jvm_path.encode("mbcs", errors="replace")
        jvm_path_w = jvm_path
        fake = self._fake_hmod
        loaded_mods = self._loader.loaded_modules
        by_handle = {mod._codebaseaddr: mod for mod in loaded_mods.values()}

        # ── GetModuleFileNameA ────────────────────────────────────────────
        try:
            self._iat_gmfa = _find_iat_entry(fake, "KERNEL32.DLL", "GetModuleFileNameA", jvm_path)
            real_gmfa = _GetModuleFileNameA_t(self._iat_gmfa.original)

            def _h_gmfa(hmod: int, buf: int | None, size: int) -> int:
                if hmod == fake:
                    if not buf or size == 0:
                        return 0
                    payload = jvm_path_bytes[: max(0, size - 1)]
                    ctypes.memmove(buf, payload, len(payload))
                    ctypes.memset(buf + len(payload), 0, 1)
                    return len(payload)
                return real_gmfa(hmod, buf, size)

            self._hook_gmfa = _GetModuleFileNameA_t(_h_gmfa)
            self._iat_gmfa.patch(ctypes.cast(self._hook_gmfa, ctypes.c_void_p).value)
            log.info("IAT hook: jvm.dll GetModuleFileNameA installed")
        except Exception as exc:
            log.warning("GetModuleFileNameA hook skipped: %s", exc)

        # ── GetModuleFileNameW ────────────────────────────────────────────
        try:
            self._iat_jvm_gmfw = _find_iat_entry(fake, "KERNEL32.DLL", "GetModuleFileNameW", jvm_path)
            real_gmfw = _GetModuleFileNameW_t(self._iat_jvm_gmfw.original)

            def _h_gmfw(hmod: int, buf: ctypes.c_wchar_p | None, size: int) -> int:
                if hmod == fake:
                    if not buf or size == 0:
                        return 0
                    src = jvm_path_w[: max(0, size - 1)]
                    ctypes.memmove(buf, (src + "\0").encode("utf-16-le"), (len(src) + 1) * 2)
                    return len(src)
                return real_gmfw(hmod, buf, size)

            self._hook_gmfw = _GetModuleFileNameW_t(_h_gmfw)
            self._iat_jvm_gmfw.patch(ctypes.cast(self._hook_gmfw, ctypes.c_void_p).value)
            log.info("IAT hook: jvm.dll GetModuleFileNameW installed")
        except Exception as exc:
            log.warning("GetModuleFileNameW hook skipped: %s", exc)

        # ── LoadLibraryA ──────────────────────────────────────────────────
        try:
            self._iat_jvm_lla = _find_iat_entry(fake, "KERNEL32.DLL", "LoadLibraryA", jvm_path)
            real_lla = _LoadLibraryA_t(self._iat_jvm_lla.original)

            def _h_jvm_lla(path: bytes | None) -> int:
                if path:
                    name = os.path.basename(path.decode("mbcs", errors="replace")).lower()
                    mod = loaded_mods.get(name)
                    if mod is not None:
                        log.info("IAT jvm.dll LoadLibraryA(%s) → memory 0x%x", name, mod._codebaseaddr)
                        return mod._codebaseaddr
                return real_lla(path)

            self._hook_jvm_lla = _LoadLibraryA_t(_h_jvm_lla)
            self._iat_jvm_lla.patch(ctypes.cast(self._hook_jvm_lla, ctypes.c_void_p).value)
            log.info("IAT hook: jvm.dll LoadLibraryA installed")
        except Exception as exc:
            log.warning("LoadLibraryA hook skipped: %s", exc)

        # ── LoadLibraryExW ────────────────────────────────────────────────
        try:
            self._iat_jvm_llexw = _find_iat_entry(fake, "KERNEL32.DLL", "LoadLibraryExW", jvm_path)
            real_llexw = _LoadLibraryExW_t(self._iat_jvm_llexw.original)

            def _h_jvm_llexw(path: str | None, hfile: int, flags: int) -> int:
                if path:
                    name = os.path.basename(path).lower()
                    mod = loaded_mods.get(name)
                    if mod is not None:
                        log.info("IAT jvm.dll LoadLibraryExW(%s, flags=0x%x) → memory 0x%x", name, flags, mod._codebaseaddr)
                        return mod._codebaseaddr
                return real_llexw(path, hfile, flags)

            self._hook_jvm_llexw = _LoadLibraryExW_t(_h_jvm_llexw)
            self._iat_jvm_llexw.patch(ctypes.cast(self._hook_jvm_llexw, ctypes.c_void_p).value)
            log.info("IAT hook: jvm.dll LoadLibraryExW installed")
        except Exception as exc:
            log.warning("LoadLibraryExW hook skipped: %s", exc)

        # ── GetProcAddress ────────────────────────────────────────────────
        try:
            self._iat_jvm_gpa = _find_iat_entry(fake, "KERNEL32.DLL", "GetProcAddress", jvm_path)
            real_gpa = _GetProcAddress_t(self._iat_jvm_gpa.original)

            def _h_jvm_gpa(hmod: int, name: bytes | None) -> int:
                mod = by_handle.get(hmod)
                if mod is not None and name:
                    try:
                        fp = mod.get_proc_addr(name.decode("ascii", errors="replace"))
                        addr = ctypes.cast(fp, ctypes.c_void_p).value
                        return addr or 0
                    except Exception:
                        log.debug("IAT jvm.dll GetProcAddress(memory, %r) → not found", name)
                        return 0
                return real_gpa(hmod, name)

            self._hook_jvm_gpa = _GetProcAddress_t(_h_jvm_gpa)
            self._iat_jvm_gpa.patch(ctypes.cast(self._hook_jvm_gpa, ctypes.c_void_p).value)
            log.info("IAT hook: jvm.dll GetProcAddress installed")
        except Exception as exc:
            log.warning("GetProcAddress (jvm) hook skipped: %s", exc)

        # ── FreeLibrary ───────────────────────────────────────────────────
        try:
            self._iat_jvm_fl = _find_iat_entry(fake, "KERNEL32.DLL", "FreeLibrary", jvm_path)
            real_fl = _FreeLibrary_t(self._iat_jvm_fl.original)

            def _h_jvm_fl(hmod: int) -> int:
                if hmod in by_handle:
                    log.debug("IAT jvm.dll FreeLibrary(memory 0x%x) → suppressed", hmod)
                    return 1
                return real_fl(hmod)

            self._hook_jvm_fl = _FreeLibrary_t(_h_jvm_fl)
            self._iat_jvm_fl.patch(ctypes.cast(self._hook_jvm_fl, ctypes.c_void_p).value)
            log.info("IAT hook: jvm.dll FreeLibrary installed")
        except Exception as exc:
            log.warning("FreeLibrary hook skipped: %s", exc)

    # ── § 4.2  _patch_iat ─────────────────────────────────────────────────

    def _patch_iat(self) -> None:
        import _jpype  # type: ignore[import]

        log.info("IAT patch: locating _jpype.pyd")
        spec = importlib.util.find_spec("_jpype")
        jpype_path = getattr(_jpype, "__file__", None) or (spec.origin if spec else None)
        if not jpype_path:
            raise RuntimeError("Cannot locate _jpype.pyd")
        log.debug("_jpype.pyd path: %s", jpype_path)

        k32 = ctypes.windll.kernel32
        k32.GetModuleHandleW.argtypes = [ctypes.c_wchar_p]
        k32.GetModuleHandleW.restype  = ctypes.c_void_p
        module_base = k32.GetModuleHandleW(str(jpype_path))
        if not module_base:
            try:
                module_base = ctypes.WinDLL(jpype_path)._handle
            except OSError:
                module_base = 0
        if not module_base:
            raise OSError(f"GetModuleHandleW failed for {jpype_path!r}: error {k32.GetLastError()}")
        log.info("_jpype.pyd base 0x%016x", module_base)
        _flush_logging()

        self._iat_llw = _find_iat_entry(module_base, "KERNEL32.DLL", "LoadLibraryW",   str(jpype_path))
        self._iat_gpa = _find_iat_entry(module_base, "KERNEL32.DLL", "GetProcAddress", str(jpype_path))
        log.info("LoadLibraryW   slot 0x%016x (orig 0x%016x)", self._iat_llw._addr, self._iat_llw.original)
        log.info("GetProcAddress slot 0x%016x (orig 0x%016x)", self._iat_gpa._addr, self._iat_gpa.original)
        _flush_logging()

        self._build_hooks()
        self._iat_llw.patch(ctypes.cast(self._hook_llw, ctypes.c_void_p).value)
        self._iat_gpa.patch(ctypes.cast(self._hook_gpa, ctypes.c_void_p).value)
        log.info("IAT hooks installed in _jpype.pyd")
        _flush_logging()

        self._patch_jvm_iat()

    def _restore_iat(self) -> None:
        for attr in ("_iat_llw", "_iat_gpa", "_iat_gmfa", "_iat_jvm_gmfw",
                     "_iat_jvm_lla", "_iat_jvm_llexw", "_iat_jvm_gpa", "_iat_jvm_fl"):
            entry = getattr(self, attr, None)
            if entry:
                try:
                    entry.restore()
                except Exception as exc:
                    log.warning("IAT restore failed for %s: %s", attr, exc)
        for attr in ("_hook_llw", "_hook_gpa", "_hook_gmfa", "_hook_gmfw",
                     "_hook_jvm_lla", "_hook_jvm_llexw", "_hook_jvm_gpa",
                     "_hook_jvm_fl", "_trampoline"):
            setattr(self, attr, None)
        log.info("IAT hooks removed — _jpype.pyd restored")

    # ── § 4.3  JPype bootstrap ────────────────────────────────────────────

    def _attach_jpype(self) -> None:
        import jpype
        import jpype.imports  # noqa: F401

        jvm_dll = str(self._loader.jdk_bin / "server" / "jvm.dll")
        log.info("jpype.startJVM(%r, ...)", jvm_dll)
        jpype.startJVM(
            jvm_dll,
            *self._jvm_flags,
            classpath=self._classpath,
            convertStrings=True,
            interrupt=True,
        )

    # ── § 4.4  Game launch ────────────────────────────────────────────────

    def _launch_game(self) -> None:
        import jpype
        log.info("Loading main class: %s", self._main_class)
        MainClass = jpype.JClass(self._main_class)
        log.info("Invoking %s.main(%d args)", self._main_class, len(self._game_args))
        if self._debug:
            log.debug("  game_args = %r", self._game_args)
        MainClass.main(self._game_args[:])
        _wait_for_java_non_daemon_threads()

    # ── § 4.5  Top-level entry ────────────────────────────────────────────

    def boot(self) -> None:
        log.info("=== JvmBootGlue.boot() starting ===")
        _enable_native_crash_log()
        _flush_logging()
        try:
            self._patch_iat()
            self._attach_jpype()
        finally:
            self._restore_iat()
        self._launch_game()
        log.info("=== JvmBootGlue.boot() returned (game exited) ===")

# § 5  PortableMCGameAdapter — extract game config from portablemc
#
# Supports portablemc 5.x PyO3 API (primary), portablemc 4.x Python API
# (secondary), and the --dry CLI fallback (last resort).
#
# The --dry fallback tries FIVE command variants and SIX parsing strategies
# so it works regardless of which portablemc version is installed.
# ══════════════════════════════════════════════════════════════════════════════

class PortableMCGameAdapter:

    def __init__(
        self,
        main_dir: Path,
        version: str,
        jdk_bin: Path,
        username: str = "Player",
        access_token: str = "0",
        extra_jvm_flags: list[str] | None = None,
    ) -> None:
        self.main_dir        = main_dir
        self.version         = version
        self.jdk_bin         = jdk_bin
        self.username        = username
        self.access_token    = access_token
        self.extra_jvm_flags = extra_jvm_flags or []

    def resolve(self) -> tuple[list[str], list[str], str, list[str]]:
        try:
            return self._resolve_via_python_api()
        except Exception as exc:
            log.warning(
                "portablemc Python API unavailable (%s); "
                "falling back to --dry parsing", exc,
            )
            return self._resolve_via_dry_run()

    # ── Primary: portablemc Python extension (in-process) ─────────────────

    def _resolve_via_python_api(self) -> tuple[list[str], list[str], str, list[str]]:
        """Try portablemc 5.x PyO3 API.  Falls back to 4.x if 5.x is absent."""
        with _CleanImportPath():
            # portablemc 5.x PyO3 (re-exports base from _portablemc.pyd)
            from portablemc import base as pmc_base  # type: ignore[import]

        jvm_path = str(self.jdk_bin / "server" / "jvm.dll")
        installer = pmc_base.Installer(
            main_dir=str(self.main_dir),
            work_dir=str(self.main_dir),
        )
        game = installer.install(
            version=self.version,
            jvm=jvm_path,
            username=self.username,
            auth_token=self.access_token,
        )
        return _PortableMCParse.extract(
            list(game.jvm_args), list(game.game_args), game.main_class
        )

    # ── Fallback: robust multi-variant --dry CLI parsing ──────────────────

    def _resolve_via_dry_run(self) -> tuple[list[str], list[str], str, list[str]]:
        """
        Run portablemc with --dry and parse any output format.

        Why five command variants?
        portablemc 4.x Python doesn't support ``--output machine`` or
        ``--work-dir``.  portablemc 5.x Rust wrapper may require them.
        We try the most compatible variant first.

        Why six parsing strategies?
        Different portablemc versions produce different output formats —
        JSON with "jvm_args" event, JSON with "game_launch" event, ANSI
        "Arguments:" text blocks, tab-delimited machine format, etc.
        """
        jvm_path = str(self.jdk_bin / "server" / "jvm.dll")
        main_dir_str = str(self.main_dir)
        safe_cwd = _safe_subprocess_cwd(self.main_dir)

        # Clean PYTHONPATH so scripts/portablemc.py doesn't shadow the package
        env = os.environ.copy()
        pp = env.get("PYTHONPATH", "")
        if pp:
            env["PYTHONPATH"] = os.pathsep.join(
                _without_local_launcher_path(pp.split(os.pathsep))
            )

        base_cmd = [sys.executable, "-m", "portablemc"]

        # Five variants, most-compatible first
        cmd_variants: list[list[str]] = [
            # 1. Default output (works with all portablemc versions)
            base_cmd + [
                "--main-dir", main_dir_str,
                "start", "--dry", "--jvm", jvm_path,
                "-u", self.username, self.version,
            ],
            # 2. Explicit human-color (portablemc 4.x compatible)
            base_cmd + [
                "--main-dir", main_dir_str, "--output", "human-color",
                "start", "--dry", "--jvm", jvm_path,
                "-u", self.username, self.version,
            ],
            # 3. Machine output (portablemc 5.x Rust / some 4.x builds)
            base_cmd + [
                "--main-dir", main_dir_str, "--output", "machine",
                "start", "--dry", "--jvm", jvm_path,
                "-u", self.username, self.version,
            ],
            # 4. With --work-dir (required by some 5.x builds)
            base_cmd + [
                "--main-dir", main_dir_str, "--work-dir", main_dir_str,
                "--output", "machine",
                "start", "--dry", "--jvm", jvm_path,
                "-u", self.username, self.version,
            ],
            # 5. Without --jvm (let portablemc find Java itself)
            base_cmd + [
                "--main-dir", main_dir_str,
                "start", "--dry",
                "-u", self.username, self.version,
            ],
        ]

        last_raw = ""
        for cmd in cmd_variants:
            try:
                proc = subprocess.run(
                    cmd,
                    capture_output=True,
                    text=True,
                    check=False,
                    cwd=safe_cwd,
                    env=env,
                    timeout=180,
                )
                raw = proc.stdout + proc.stderr
                log.debug(
                    "dry-run variant: %s ...\n--- output (%d chars) ---\n%s\n---",
                    " ".join(cmd[:6]), len(raw), raw[:5000],
                )
                last_raw = raw
                if not raw.strip():
                    continue

                result = _parse_portablemc_dry_output(raw)
                if result is not None:
                    jvm_flags, classpath, main_class, game_args = result
                    if classpath or main_class:
                        return result

            except subprocess.TimeoutExpired:
                log.warning("dry-run variant timed out: %s", " ".join(cmd[:4]))
            except Exception as exc:
                log.warning("dry-run variant exception (%s): %s", " ".join(cmd[:4]), exc)

        log.error(
            "All dry-run variants failed.\n"
            "Install portablemc:  python -m pip install portablemc\n"
            "Last raw output:\n%s",
            last_raw[:8000],
        )
        raise RuntimeError(
            "dry-run produced no parseable Java command.\n"
            "Ensure portablemc is installed: python -m pip install portablemc\n"
            f"Last output:\n{last_raw[:2000]}"
        )

    # ── Static helpers (kept for backwards compatibility) ─────────────────

    @staticmethod
    def _extract_args(
        jvm_args: list[str],
        game_args: list[str],
        main_class: str,
    ) -> tuple[list[str], list[str], str, list[str]]:
        return _PortableMCParse.extract(jvm_args, game_args, main_class)

    @staticmethod
    def _extract_args_from_java_command(
        args: list[str],
    ) -> tuple[list[str], list[str], str, list[str]]:
        """Parse a flat java command line (java.exe stripped)."""
        return _PortableMCParse.split_flat(args)

# § 6  Top-level entry point
# ══════════════════════════════════════════════════════════════════════════════

def launch_minecraft(
    jdk_bin: Path,
    main_dir: Path,
    version: str = "fabric:latest",
    username: str = "Player",
    access_token: str = "0",
    extra_jvm_flags: list[str] | None = None,
    extra_game_args: list[str] | None = None,
    debug: bool = False,
    memory_resident: bool = False,
) -> None:
    """
    Complete in-process Minecraft launcher.  No ``.exe`` is spawned.

    memory_resident=False (option 4)
        Load jvm.dll via the normal OS loader (jpype.startJVM with the real
        jvm.dll path).  Simple and reliable.  Requires no IAT hooks.

    memory_resident=True (option 5)
        Map jvm.dll and all JDK deps into RAM via JvmMemoryLoader, then boot
        JPype through IAT-hooked startJVM.  Maximum in-process isolation.
    """
    logging.basicConfig(
        level=logging.DEBUG if debug else logging.INFO,
        format="%(asctime)s %(levelname)-8s %(name)s — %(message)s",
    )

    # ── Step 1: optionally load JVM from memory ───────────────────────────
    loader = None
    if memory_resident:
        from jvm_memory_loader import JvmMemoryLoader  # type: ignore[import]
        log.info("Loading JVM from memory (jdk_bin=%s)", jdk_bin)
        loader = JvmMemoryLoader(jdk_bin=jdk_bin, debug=debug).load()
        if not loader.verify():
            raise RuntimeError("JNI_CreateJavaVM verification failed — aborting")

    # ── Step 2: resolve Minecraft game configuration ──────────────────────
    log.info("Resolving Minecraft version: %r", version)
    adapter = PortableMCGameAdapter(
        main_dir=main_dir,
        version=version,
        jdk_bin=jdk_bin,
        username=username,
        access_token=access_token,
        extra_jvm_flags=extra_jvm_flags,
    )
    jvm_flags, classpath, main_class, game_args = adapter.resolve()

    if extra_jvm_flags:
        jvm_flags = list(jvm_flags) + list(extra_jvm_flags)
    if extra_game_args:
        game_args = list(game_args) + list(extra_game_args)

    if not main_class:
        raise RuntimeError("portablemc did not provide a Minecraft main class")
    if not classpath:
        raise RuntimeError("portablemc did not provide a Minecraft classpath")

    if memory_resident:
        jdk_home = str(jdk_bin.parent)
        boot_lib = str(jdk_bin)
        strip_prefixes = ("-Djava.home=", "-Djdk.home=", "-Dsun.boot.library.path=", "-Djava.library.path=")
        jvm_flags = [
            f"-Djava.home={jdk_home}",
            f"-Djdk.home={jdk_home}",
            f"-Dsun.boot.library.path={boot_lib}",
            f"-Djava.library.path={boot_lib}",
            *[f for f in jvm_flags if not any(f.startswith(p) for p in strip_prefixes)],
        ]
        log.info("Memory JVM home forced: %s", jdk_home)

    log.info(
        "Game config: main_class=%s  jvm_flags=%d  classpath=%d  game_args=%d",
        main_class, len(jvm_flags), len(classpath), len(game_args),
    )

    # ── Step 3: boot + launch ─────────────────────────────────────────────
    if not memory_resident:
        # Option 4: OS-loader JVM (standard jpype.startJVM, no IAT tricks)
        import jpype
        import jpype.imports  # noqa: F401
        jvm_dll = str(jdk_bin / "server" / "jvm.dll")
        log.info("Starting in-process JVM via OS loader: %s", jvm_dll)
        jpype.startJVM(
            jvm_dll,
            *jvm_flags,
            classpath=classpath,
            convertStrings=True,
            interrupt=True,
        )
        log.info("Loading main class: %s", main_class)
        jpype.JClass(main_class).main(game_args[:])
        _wait_for_java_non_daemon_threads()
        return

    # Option 5: memory-resident JVM (IAT-hooked jpype.startJVM)
    JvmBootGlue(
        loader=loader,
        jvm_flags=jvm_flags,
        classpath=classpath,
        main_class=main_class,
        game_args=game_args,
        debug=debug,
    ).boot()
